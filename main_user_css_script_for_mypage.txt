[*# GitHub ファイル埋め込み]
 GitHub 埋め込みは、Cosense の CSP 制約を避けるため、Tampermonkey 用の `cosense_github_embed_tampermonkey.user.js` 側で CSS と JavaScript を管理する。
 マイページの UserScript から GitHub Raw への `fetch()` は行わない。

[*# X（Twitter）ポスト埋め込み]
 X（Twitter）ポスト埋め込みは、X の Syndication API から投稿データを取得し、iframe を使わない静的なカードとして描画する Tampermonkey 用の `cosense_x_post_embed_tampermonkey.user.js` 側で CSS と JavaScript を管理する。
 ポスト URL だけの行を対象とし、通常はリプライ元を含め、`?conversation=none` の指定時だけリプライ元を非表示にする。

[*# 表（テーブル）]
 各列で本文幅を均等に分け、列数が多い場合だけ文字と余白を少し小さくする方法を実装している
 	（デフォルトの場合、本文表示範囲からはみ出してしまう場合があるという問題があったので、これを防ぐ実装にしている）
 		[**** [https://i.gyazo.com/ecf612f1c1d3fcccfe51119fd46a09bb.png]]
		これにより、例えば、4列の場合は各列が本文幅の約4分の1、6列なら約6分の1となり、固定 16em のように列数に比例して表全体が横へ伸びなくなる
code:style.css
 /*
  * Zenn 風の表表示。
  *
  * 本文描画エリアの幅を各列で均等に分け、
  * 列数が増えた場合は文字と余白を少し小さくする。
  */
 .table-block {
   width: 100%;
   max-width: 100%;
	
   font-size: 0.95em;
   line-height: 1.5;
 }
	
	
 /*
  * 表の行。
  *
  * 編集中の行へレイアウト変更を適用すると、
  * Cosense のカーソル移動へ影響する可能性があるため、
  * 非編集行だけを対象にする。
  */
 .line:not(.cursor-line) .table-block-row {
   display: flex !important;
   align-items: stretch !important;
	
   /* インデントした表は Cosense が行へ margin-left を付けるため、auto にして本文幅からはみ出さないようにする */
   width: auto !important;
   max-width: 100% !important;
   height: auto !important;
 }
	
	
 /*
  * 本文幅をすべてのセルで均等に分ける。
  */
 .line:not(.cursor-line) .table-block .cell {
   display: flex !important;
   align-items: flex-start;
	
   flex: 1 1 0 !important;
   width: 0 !important;
   min-width: 0 !important;
   max-width: none !important;
	
   box-sizing: border-box;
   vertical-align: top;
 }
	
	
 /*
  * セル内テキストを自動で折り返す。
  */
 .line:not(.cursor-line) .table-block .cell-text {
   display: block;
   box-sizing: border-box;
	
   width: 100%;
   min-width: 0;
   max-width: 100%;
	
   white-space: normal;
   overflow-wrap: anywhere;
   word-break: normal;
   text-wrap: pretty;
	
   vertical-align: top;
 }
	
	
 /*
  * 4列以上では文字と余白を少しコンパクトにする。
  */
 .line:not(.cursor-line)
 .table-block-row:has(> .cell:nth-child(4))
 .cell {
   padding: 0.4rem;
 }
	
 .line:not(.cursor-line)
 .table-block-row:has(> .cell:nth-child(4))
 .cell-text {
   font-size: 0.9em;
   line-height: 1.45;
 }
	
	
 /*
  * 6列以上ではさらにコンパクトにする。
  */
 .line:not(.cursor-line)
 .table-block-row:has(> .cell:nth-child(6))
 .cell {
   padding: 0.3rem;
 }
	
 .line:not(.cursor-line)
 .table-block-row:has(> .cell:nth-child(6))
 .cell-text {
   font-size: 0.82em;
   line-height: 1.4;
 }
	
	
 /* table:テーブル名 の表示を控えめにする */
 .table-block .table-block-start {
   color: #65717b;
   background-color: transparent;
   padding: 0.2em 0.1em 0.35em;
   font-size: 0.85em;
   font-weight: 600;
 }
	
 .table-block .table-block-start a {
   color: #65717b;
 }
	
	
 /* Cosense 標準の列ごとの背景色を白にそろえる */
 .table-block .cell:nth-child(odd),
 .table-block .cell:nth-child(even) {
   color: inherit;
   background-color: #fff;
 }
	
	
 /* 各セルの余白と境界線 */
 .table-block .cell {
   box-sizing: border-box;
   padding: 0.5rem;
	
   border-right: solid 1px #d6e3ed;
   border-bottom: solid 1px #d6e3ed;
 }
	
 .table-block .cell:first-child {
   border-left: solid 1px #d6e3ed;
 }
	
	
 /* 4列以上では、上の通常余白を上書きする */
 .line:not(.cursor-line)
 .table-block-row:has(> .cell:nth-child(4))
 .cell {
   padding: 0.4rem;
 }
	
	
 /* 6列以上では、上の4列用余白を上書きする */
 .line:not(.cursor-line)
 .table-block-row:has(> .cell:nth-child(6))
 .cell {
   padding: 0.3rem;
 }
	
	
 /* スマホ幅では余白をさらに詰めて、1 列あたりの文字数を確保する */
 @media (max-width: 767px) {
   .line:not(.cursor-line) .table-block .cell {
     padding: 0.35rem;
   }
 
   .line:not(.cursor-line)
   .table-block-row:has(> .cell:nth-child(4))
   .cell {
     padding: 0.25rem;
   }
 }
	
	
 /* 先頭行を見出し行として表示する */
 .line:has(.table-block-start) + .line .table-block .cell,
 .section-title + .line .table-block .cell {
   border-top: solid 1px #d6e3ed;
   background-color: #edf2f7;
   font-weight: 700;
 }
	
	
 /* 編集中の行がセル背景に隠れないようにする */

   background-color: #f0f7ff;

[#* ページ内目次の自動生成]
	from: [/daiiz/ページの見出しを作るUserScript]
 `[*# 見出し]` のように `#` を含む太字だけの行を見出しとして、`*` の数に応じた階層付きで一覧にする
 `#` を含む見出しがないページでは、`[** 見出し]` のように太字だけの行を使う
code:script.js
 /*
  * ページ内の見出しを集める。
  * level は * の数、depth は最上位の見出しを 0 とした階層の深さ。
  */
 function getHeadings () {
   const candidates = []
   for (const line of scrapbox.Page.lines ?? []) {
     if (line.title || !line.nodes) continue
     const node = getHeadingNode(line.nodes)
     if (!node) continue
     const image = getIconUrl(node.children)
     const text = renderPlainText(node.children, {noIcon: true}).trim()
     const textWithIcons = renderPlainText(node.children, {noIcon: false}).trim()
     candidates.push({
       id: line.id,
       level: node.type === 'strong' ? 1 : node.unit.strong,
       hasHash: node.type === 'deco' && node.unit.deco.includes('#'),
       text: text || textWithIcons,
       label: image ? text : textWithIcons,
       image
     })
   }
   const headings = candidates.some(heading => heading.hasHash)
     ? candidates.filter(heading => heading.hasHash)
     : candidates
   const top = Math.max(...headings.map(heading => heading.level))
   for (const heading of headings) heading.depth = Math.min(top - heading.level, 3)
   return headings
 }
 
 // 行の中身が太字の装飾（[* ...] や [[...]]）1 つだけのとき、その装飾を返す
 function getHeadingNode (nodes) {
   const list = (Array.isArray(nodes) ? nodes : [nodes])
     .filter(node => typeof node !== 'string' || node.trim() !== '')
   if (list.length !== 1) return null
   const [node] = list
   if (node.type === 'indent') return getHeadingNode(node.children)
   if (node.type === 'strong') return node
   if (node.type === 'deco' && node.unit.strong > 0) return node
   return null
 }
 
 function renderPlainText (node, options) {
   if (node == null) return ''
   if (Array.isArray(node)) return node.map(child => renderPlainText(child, options)).join('')
   if (typeof node === 'string') return node
   switch (node.type) {
     case 'icon':
     case 'strong-icon':
       return options.noIcon ? ' ' : node.unit.page
     case 'hashTag':
       return ''
   }
   return renderPlainText(node.children, options)
 }
 
 function getIconUrl (node) {
   if (node == null || typeof node === 'string') return null
   if (Array.isArray(node)) {
     for (const child of node) {
       const url = getIconUrl(child)
       if (url) return url
     }
     return null
   }
   if (node.type === 'icon' || node.type === 'strong-icon') {
     const project = node.unit.project || scrapbox.Project.name
     return `/api/pages/${encodeURIComponent(project)}/${encodeURIComponent(node.unit.page)}/icon`
   }
   return getIconUrl(node.children)
 }
 
 // 行リンク（#行ID）で移動する。すでに同じ行リンクを開いているときは、その行までスクロールする
 function jumpToLine (id) {
   if (location.hash !== `#${id}`) {
     location.hash = id
     return
   }
   const line = document.getElementById(`L${id}`)
   if (line) window.scrollTo({top: line.getBoundingClientRect().top + window.scrollY - 100, behavior: 'smooth'})
 }
 
 scrapbox.PageMenu.addMenu({
   title: '見出し',
   image: 'https://gyazo.com/da0408108eda84e56a587630be5e4524/raw',
   onClick: () => {
     const menu = scrapbox.PageMenu('見出し')
     menu.removeAllItems()
     const headings = getHeadings()
     if (headings.length === 0) {
       menu.addItem({title: '見出しはありません', onClick: () => {}})
       return
     }
     for (const heading of headings) {
       menu.addItem({
         title: '　'.repeat(heading.depth) + heading.label,
         image: heading.image ?? undefined,
         onClick: () => jumpToLine(heading.id)
       })
     }
   }
 })

[#* Emoji (絵文字)]
	from: [/yutaro/emoji selector]
 `:` に続けて名前を入力すると、アイコン（絵文字）の候補を表示する
	↑↓ または Ctrl+P / Ctrl+N で選択、Enter または Tab で挿入、Esc で閉じる。`:名前:` と入力しても挿入できる
	英数字の直後の `:`（URL の `https:` や時刻の `12:30` など）では候補を出さない
	アイコンは最初に `:` を入力したときに読み込み、ページを移動しても使い回す
code:script.js
 // ブロック内の変数は、ほかの機能の変数と衝突しない
 {
   const MAX_ITEMS = 30
   const NAME_PATTERN = /^[\w\s+-]+$/
   const projectName = scrapbox.Project.name
   const emojis = []
   const knownNames = new Set()
   const loadedProjects = new Set()
   let loading = null
 
   const iconUrl = (project, title) =>
     `/api/pages/${encodeURIComponent(project)}/${encodeURIComponent(title)}/icon`
 
   // アイコン画像を持ち、タイトルが英数字のページを、続きがなくなるまでたどってすべて取得する
   const fetchIconTitles = async project => {
     const titles = []
     let followingId = ''
     do {
       const query = followingId ? `?followingId=${encodeURIComponent(followingId)}` : ''
       const response = await fetch(`/api/pages/${encodeURIComponent(project)}/search/titles${query}`, {credentials: 'same-origin'})
       if (!response.ok) throw new Error(`${response.status} ${response.statusText}`)
       for (const page of await response.json()) {
         if (page.image && NAME_PATTERN.test(page.title)) titles.push(page.title)
       }
       followingId = response.headers.get('X-Following-Id') ?? ''
     } while (followingId)
     return titles
   }
 
   const loadEmojis = async project => {
     if (loadedProjects.has(project)) return
     const titles = await fetchIconTitles(project)
     loadedProjects.add(project)
     for (const name of titles) {
       if (knownNames.has(name)) continue
       knownNames.add(name)
       emojis.push({
         name,
         path: project === projectName ? name : `/${project}/${name}`,
         icon: iconUrl(project, name)
       })
     }
   }
 
   const loadProjectEmojis = () => {
     loading ??= loadEmojis(projectName).catch(error => {
       loading = null
       console.error('[Cosense Zenn Style] 絵文字の候補を読み込めませんでした', error)
     })
     return loading
   }
 
   // 並び順: 完全一致 → 前方一致 → 単語の前方一致 → 部分一致 → 文字を順に含む
   const rank = (name, query) => {
     const target = name.toLowerCase()
     if (target === query) return 0
     if (target.startsWith(query)) return 1
     if (target.split(/[\s_+-]+/).some(word => word.startsWith(query))) return 2
     if (target.includes(query)) return 3
     let index = 0
     for (const char of target) {
       if (char === query[index]) index++
     }
     return index === query.length ? 4 : -1
   }
 
   const search = text => {
     const query = text.toLowerCase()
     return emojis
       .map(emoji => ({emoji, score: rank(emoji.name, query)}))
       .filter(({score}) => score >= 0)
       .sort((a, b) => a.score - b.score || a.emoji.name.length - b.emoji.name.length || a.emoji.name.localeCompare(b.emoji.name))
       .slice(0, MAX_ITEMS)
       .map(({emoji}) => emoji)
   }
 
   const menu = document.createElement('div')
   menu.className = 'zs-emoji-menu'
   menu.setAttribute('role', 'listbox')
   menu.setAttribute('aria-label', '絵文字の候補')
   menu.hidden = true
   document.body.append(menu)
 
   // 候補を押してもエディターからフォーカスが外れないようにする
   menu.addEventListener('mousedown', event => {
     event.preventDefault()
     event.stopPropagation()
   })
 
   let active = false // 「:」を入力して、名前を入力している途中か
   let query = ''
   let matches = []
   let selected = 0
 
   const close = () => {
     active = false
     query = ''
     matches = []
     menu.hidden = true
   }
 
   const select = index => {
     selected = (index + matches.length) % matches.length
     for (const [i, item] of [...menu.children].entries()) {
       item.classList.toggle('is-selected', i === selected)
       item.setAttribute('aria-selected', String(i === selected))
     }
     const item = menu.children[selected]
     if (!item) return
     if (item.offsetTop < menu.scrollTop) {
       menu.scrollTop = item.offsetTop
     } else if (item.offsetTop + item.offsetHeight > menu.scrollTop + menu.clientHeight) {
       menu.scrollTop = item.offsetTop + item.offsetHeight - menu.clientHeight
     }
   }
 
   // カーソルの下（下に入らないときは上）に、画面からはみ出さないように表示する
   const place = () => {
     const caret = document.querySelector('.cursor-container .cursor') ?? document.getElementById('text-input')
     if (!caret) return
     const rect = caret.getBoundingClientRect()
     menu.hidden = false
     const width = menu.offsetWidth
     const height = menu.offsetHeight
     const left = Math.max(8, Math.min(rect.left, document.documentElement.clientWidth - width - 8))
     let top = rect.bottom + 4
     if (top + height > window.innerHeight - 8 && rect.top - height - 4 > 48) top = rect.top - height - 4
     menu.style.left = `${Math.round(left + window.scrollX)}px`
     menu.style.top = `${Math.round(top + window.scrollY)}px`
   }
 
   const insert = emoji => {
     const input = document.getElementById('text-input')
     const length = query.length + 1 // 「:」と、その後に入力した名前の文字数
     close()
     if (!input || !emoji) return
     input.focus()
     setTimeout(() => {
       for (let i = 0; i < length; i++) {
         const event = document.createEvent('Events')
         event.initEvent('keydown', true, true)
         event.keyCode = event.which = 8 // Backspace
         input.dispatchEvent(event)
       }
       document.execCommand('insertText', false, `[${emoji.path}.icon]`)
     }, 0)
   }
 
   const render = () => {
     if (!active || query === '') {
       menu.hidden = true
       return
     }
     matches = search(query)
     if (matches.length === 0) {
       menu.hidden = true
       return
     }
     selected = Math.min(selected, matches.length - 1)
     menu.replaceChildren(...matches.map((emoji, index) => {
       const item = document.createElement('div')
       item.className = 'zs-emoji-item'
       item.setAttribute('role', 'option')
       item.setAttribute('aria-selected', String(index === selected))
       item.classList.toggle('is-selected', index === selected)
       const image = document.createElement('img')
       image.src = emoji.icon
       image.alt = ''
       image.loading = 'lazy'
       // アイコン画像を読み込めないときは、名前の頭文字を代わりに表示する
       image.addEventListener('error', () => {
         const placeholder = document.createElement('span')
         placeholder.className = 'zs-emoji-placeholder'
         placeholder.textContent = emoji.name.slice(0, 1).toUpperCase()
         image.replaceWith(placeholder)
       }, {once: true})
       const name = document.createElement('span')
       name.textContent = `:${emoji.name}:`
       item.append(image, name)
       item.addEventListener('click', () => insert(emoji))
       return item
     }))
     menu.scrollTop = 0
     place()
   }
 
   // カーソル直前の 1 文字（行頭なら空文字）を、カーソルと文字の表示位置から求める
   const charBeforeCursor = line => {
     const caret = document.querySelector('.cursor-container .cursor')
     if (!caret) return ''
     const rect = caret.getBoundingClientRect()
     const middle = rect.top + rect.height / 2
     let before = ''
     for (const char of line.querySelectorAll('span.char-index')) {
       const box = char.getBoundingClientRect()
       if (box.width === 0 || box.top > middle || box.bottom < middle) continue
       if (Math.abs(box.right - rect.left) <= 2) before = char.dataset.char ?? char.textContent
     }
     return before
   }
 
   // 行頭・空白・日本語などの直後の「:」だけで候補を出す
   const canStart = () => {
     if (scrapbox.Layout !== 'page') return false
     const line = document.querySelector('.lines .line.cursor-line')
     if (!line || line.querySelector('.code-block')) return false
     const before = charBeforeCursor(line)
     return before === '' || /\s/.test(before) || !/[\x21-\x7e]/.test(before)
   }
 
   const consume = event => {
     event.preventDefault()
     event.stopPropagation()
   }
 
   document.addEventListener('keydown', event => {
     if (!event.isTrusted || event.target?.id !== 'text-input') return
     if (event.isComposing || event.key === 'Process') return
     const {key} = event
 
     if (!active) {
       if (key === ':' && canStart()) {
         active = true
         query = ''
         selected = 0
         loadProjectEmojis().then(() => requestAnimationFrame(render))
       }
       return
     }
 
     if (!menu.hidden && matches.length > 0) {
       const down = key === 'ArrowDown' || (event.ctrlKey && key === 'n')
       const up = key === 'ArrowUp' || (event.ctrlKey && key === 'p')
       if (down || up) {
         consume(event)
         select(selected + (down ? 1 : -1))
         return
       }
       if (key === 'Enter' || key === 'Tab') {
         consume(event)
         insert(matches[selected])
         return
       }
       if (key === 'Escape') {
         consume(event)
         close()
         return
       }
     }
 
     if (key === ':') {
       const emoji = emojis.find(item => item.name === query)
       if (emoji) {
         consume(event)
         insert(emoji)
       } else {
         close()
       }
       return
     }
 
     if (key === 'Backspace') {
       if (query === '') {
         close()
       } else {
         query = query.slice(0, -1)
         selected = 0
         requestAnimationFrame(render)
       }
       return
     }
 
     if (key.length === 1 && NAME_PATTERN.test(key) && !event.ctrlKey && !event.metaKey && !event.altKey) {
       if (key === ' ' && query === '') {
         close()
         return
       }
       query += key
       selected = 0
       requestAnimationFrame(render)
       return
     }
 
     if (['Shift', 'Control', 'Alt', 'Meta', 'CapsLock'].includes(key)) return
     close()
   }, true)
 
   // 候補の外をクリックしたときや、ページを移動したときは閉じる
   document.addEventListener('pointerdown', event => {
     if (active && !menu.contains(event.target)) close()
   }, true)
   scrapbox.on('page:changed', close)
 
   scrapbox.PageMenu.addMenu({
     title: 'emoji',
     image: 'https://i.gyazo.com/49db853b2f42e1e26d07e3f982eae2f4.png'
   })
   scrapbox.PageMenu('emoji').addItem({
     title: 'load emojis from /emoji',
     onClick: async () => {
       try {
         await loadProjectEmojis() // 同じ名前は自分のプロジェクトのアイコンを優先する
         await loadEmojis('emoji')
       } catch (error) {
         console.error('[Cosense Zenn Style] /emoji の絵文字を読み込めませんでした', error)
       }
     }
   })
 }
